% proof that E[XY] = E[X]E[Y], just before "V.A. MULTINOMIALE"): product sigma-algebras and

% absolutely continuous random vectors, and an application of Fubini-Tonelli.

This chapter builds, from two measure spaces, a measure space on their Cartesian product:
first a $\sigma$-algebra, generated by the measurable rectangles, then a measure, which gives
a rectangle the product of the measures of its sides. In probability, a pair of random
variables is a random vector, and its law is a measure on the plane: the two random variables
are independent exactly when this law is the product of their laws. Tonelli's and Fubini's
theorems, which turn an integral with respect to a product measure into two iterated simple
integrals, are the last topic of measure theory in the notes; we use them to find the
marginal densities of an absolutely continuous random vector, to characterise independence
through densities, and to prove that $\mathbb E[XY]=\mathbb E[X]\,\mathbb E[Y]$ for
independent integrable $X$ and $Y$.

\section{Product measure}\label{sec:pm-product}

Let $(X_1,\mathcal A_1,\mu_1)$ and $(X_2,\mathcal A_2,\mu_2)$ be measure spaces
(\cref{def:ms-finite}). We want to give $X_1\times X_2$ the structure of a measure space.

\emph{Step 1: we define a $\sigma$-algebra.}

\begin{definition}[Measurable rectangle]\label{def:pm-rectangle}
A \emph{measurable rectangle} is a subset of $X_1\times X_2$ of the form $A\times B$, where
$A\in\mathcal A_1$ and $B\in\mathcal A_2$.
\end{definition}

\begin{remark}\label{rem:pm-rectangles}
The family of measurable rectangles is not a $\sigma$-algebra in general; it is not even an
algebra. For example, let $A_1,A_2\in\mathcal A_1$ and $B_1,B_2\in\mathcal A_2$ be non-empty,
with $A_1\cap A_2=\emptyset$ and $B_1\cap B_2=\emptyset$ (two rectangles placed diagonally,
as in the picture of the notes). Then $(A_1\times B_1)\cup(A_2\times B_2)$ is not a rectangle: if it
were $C\times D$, then $C\supseteq A_1\cup A_2$ and $D\supseteq B_1\cup B_2$, so $C\times D$
would contain the points $(a_1,b_2)$ with $a_1\in A_1$ and $b_2\in B_2$, which lie in neither
rectangle.\footnote{The handwritten notes state this without ``in general''; corrected here:
if $\mathcal A_1=\{\emptyset,X_1\}$ and $\mathcal A_2=\{\emptyset,X_2\}$, the measurable
rectangles are $\emptyset$ and $X_1\times X_2$, which do form a $\sigma$-algebra. The notes
only draw the two rectangles; the argument is added here.}
\end{remark}

\begin{definition}[Product $\sigma$-algebra]\label{def:pm-product-sigma}
The \emph{product $\sigma$-algebra} of $\mathcal A_1$ and $\mathcal A_2$, denoted
$\mathcal A_1\otimes\mathcal A_2$, is the $\sigma$-algebra generated by the measurable
rectangles (\cref{def:sa-generated}):
\[
  \mathcal A_1\otimes\mathcal A_2=\sigma\big(\{A\times B:\ A\in\mathcal A_1,\ B\in\mathcal A_2\}\big).
\]
\end{definition}

Consider the canonical projections
\[
  \pi_1:X_1\times X_2\to X_1,\quad (x,y)\mapsto x,\qquad\qquad
  \pi_2:X_1\times X_2\to X_2,\quad (x,y)\mapsto y .
\]

\begin{proposition}\label{prop:pm-projections}
The product $\sigma$-algebra is the $\sigma$-algebra generated by the projections
(\cref{def:mf-generated}):
\[
  \mathcal A_1\otimes\mathcal A_2=\sigma(\pi_1,\pi_2)
  =\sigma\big(\{\pi_1^{-1}(A),\ \pi_2^{-1}(B):\ A\in\mathcal A_1,\ B\in\mathcal A_2\}\big).
\]
\end{proposition}

\begin{proof}
``$\supseteq$''. For $A\in\mathcal A_1$ and $B\in\mathcal A_2$,
\[
  \pi_1^{-1}(A)=A\times X_2\in\mathcal A_1\otimes\mathcal A_2,\qquad
  \pi_2^{-1}(B)=X_1\times B\in\mathcal A_1\otimes\mathcal A_2,
\]
since $A\times X_2$ (a vertical strip, in the picture of the notes) and $X_1\times B$ (a
horizontal strip) are measurable rectangles. So $\mathcal A_1\otimes\mathcal A_2$ is a
$\sigma$-algebra containing the generators of $\sigma(\pi_1,\pi_2)$, and
$\sigma(\pi_1,\pi_2)\subseteq\mathcal A_1\otimes\mathcal A_2$.

``$\subseteq$''. Let $A\times B$ be a measurable rectangle, $A\in\mathcal A_1$,
$B\in\mathcal A_2$. It is the intersection of the two strips:
\[
  A\times B=(A\times X_2)\cap(X_1\times B)=\pi_1^{-1}(A)\cap\pi_2^{-1}(B)\in\sigma(\pi_1,\pi_2).
\]
So $\sigma(\pi_1,\pi_2)$ is a $\sigma$-algebra containing the measurable rectangles, and
$\mathcal A_1\otimes\mathcal A_2\subseteq\sigma(\pi_1,\pi_2)$.
\end{proof}

Hence $(X_1\times X_2,\mathcal A_1\otimes\mathcal A_2)$ is a measurable space.

\emph{Step 2: we now define a measure.} Let $E\in\mathcal A_1\otimes\mathcal A_2$; we define
its sections.

\begin{definition}[Sections]\label{def:pm-sections}
Let $E\in\mathcal A_1\otimes\mathcal A_2$. For every $x\in X_1$ and every $y\in X_2$, the
\emph{sections} of $E$ at $x$ and at $y$ are
\[
  E^2_x=\{y\in X_2:(x,y)\in E\}\subseteq X_2,\qquad
  E^1_y=\{x\in X_1:(x,y)\in E\}\subseteq X_1 .
\]
\end{definition}

    \cref{rem:pm-borel-plane}).}
  \label{fig:pm-sections}
\end{figure}

\begin{proposition}\label{prop:pm-sections}
If $\mu_1$ and $\mu_2$ are $\sigma$-finite (\cref{def:ms-finite}), then for every
$E\in\mathcal A_1\otimes\mathcal A_2$:
\begin{enumerate}[label=\arabic*)]
  \item $E^2_x\in\mathcal A_2$ for every $x\in X_1$, and $E^1_y\in\mathcal A_1$ for every
    $y\in X_2$;
  \item the functions
    \[
      X_1\to[0,+\infty],\quad x\mapsto\mu_2(E^2_x),\qquad\qquad
      X_2\to[0,+\infty],\quad y\mapsto\mu_1(E^1_y)
    \]
    are measurable;
  \item $\displaystyle\int_{X_1}\mu_2(E^2_x)\,\dd\mu_1(x)=\int_{X_2}\mu_1(E^1_y)\,\dd\mu_2(y)$.
\end{enumerate}
\end{proposition}

We do not prove the proposition, but we verify it on rectangles. Let
$E=A\times B\in\mathcal A_1\otimes\mathcal A_2$, with $A\in\mathcal A_1$ and $B\in\mathcal A_2$.
\begin{enumerate}[label=\arabic*)]
  \item The sections are
    \[
      E^2_x=\begin{cases}B&\text{if }x\in A,\\ \emptyset&\text{if }x\notin A,\end{cases}
      \qquad\in\mathcal A_2,
      \qquad\qquad
      E^1_y=\begin{cases}A&\text{if }y\in B,\\ \emptyset&\text{if }y\notin B,\end{cases}
      \qquad\in\mathcal A_1 .
    \]
  \item $x\mapsto\mu_2(E^2_x)=\mu_2(B)\1_A(x)$ and $y\mapsto\mu_1(E^1_y)=\mu_1(A)\1_B(y)$ are
    non-negative simple functions, hence measurable.
  \item By the integral of a non-negative simple function (\cref{def:in-simple-integral}),
    \[
      \int_{X_1}\mu_2(E^2_x)\,\dd\mu_1(x)=\int_{X_1}\mu_2(B)\1_A(x)\,\dd\mu_1(x)
      =\mu_2(B)\int_{X_1}\1_A\,\dd\mu_1=\mu_2(B)\mu_1(A),
    \]
    and in the same way $\int_{X_2}\mu_1(E^1_y)\,\dd\mu_2(y)=\mu_1(A)\mu_2(B)$, with the
    convention $0\cdot\infty=0$.
\end{enumerate}
For the proof see \citet[Theorem~18.1(i) and the proof of Theorem~18.2]{billingsley} or
\citet[Chapter~8, up to Theorem~8.6]{rudin-rca}. Part 1 does not need $\sigma$-finiteness
(\exref{ex:pm-product-measure}); part 3 does (\exref{ex:pm-hypotheses}), and part 2 can fail
too without it.\footnote{These observations are added to the handwritten notes.}

\begin{theorem}[Product measure]\label{thm:pm-product-measure}
Let $\mu_1$ and $\mu_2$ be $\sigma$-finite. For every $E\in\mathcal A_1\otimes\mathcal A_2$ set
\[
  \mu_1\otimes\mu_2(E)=\int_{X_1}\mu_2(E^2_x)\,\dd\mu_1(x)=\int_{X_2}\mu_1(E^1_y)\,\dd\mu_2(y).
\]
Then $\mu_1\otimes\mu_2$ is a $\sigma$-finite measure on
$(X_1\times X_2,\mathcal A_1\otimes\mathcal A_2)$, called the \emph{product measure} of
$\mu_1$ and $\mu_2$. Moreover
\[
  \mu_1\otimes\mu_2(A\times B)=\mu_1(A)\mu_2(B)\qquad\forall A\in\mathcal A_1,\ B\in\mathcal A_2 .
\]
Finally, if $\lambda$ is a measure on $(X_1\times X_2,\mathcal A_1\otimes\mathcal A_2)$ such
that $\lambda(A\times B)=\mu_1(A)\mu_2(B)$ for all $A\in\mathcal A_1$ and $B\in\mathcal A_2$,
then $\lambda=\mu_1\otimes\mu_2$.
\end{theorem}

The notes state the theorem without proof. The two integrals are equal by
\cref{prop:pm-sections}, and the value on rectangles is part~3 of the verification above.
For the proof see \citet[Theorem~18.2]{billingsley}; \citet[Theorem~8.6 and the definition
that follows it]{rudin-rca} constructs the product measure but does not state its uniqueness,
for which see Billingsley. \exref{ex:pm-product-measure} derives the theorem from
\cref{prop:pm-sections}, the uniqueness coming from Dynkin's $\pi$--$\lambda$ theorem.

\begin{remark}[Null sets have null sections]\label{rem:pm-null-sections}
Let $\mu_1$ and $\mu_2$ be $\sigma$-finite. If $E\in\mathcal A_1\otimes\mathcal A_2$ and
$\mu_1\otimes\mu_2(E)=0$, then
\[
  \mu_2(E^2_x)=0\quad\text{for $\mu_1$-a.e.\ }x\in X_1,\qquad
  \mu_1(E^1_y)=0\quad\text{for $\mu_2$-a.e.\ }y\in X_2 .
\]
Indeed $\int_{X_1}\mu_2(E^2_x)\,\dd\mu_1(x)=0$, and a non-negative measurable function with
integral $0$ vanishes a.e.\ (\cref{prop:in-ae}); the same for $y$. The notes illustrate this
in $\mathbb R^2$ with two segments of zero area. On a segment of the diagonal, every section
$E^2_x$ is a single point, of length $0$. On a vertical segment $E=\{x_0\}\times[a,b]$, the
section $E^2_{x_0}=[a,b]$ has positive length, but this happens at the single point $x_0$, a
set of length $0$.
\end{remark}

\begin{example}[The plane]\label{rem:pm-borel-plane}
Let us work in $\mathbb R^2=\mathbb R\times\mathbb R$. The Borel $\sigma$-algebra of
$\mathbb R^2$ (\cref{def:sa-borel}) coincides with the product $\sigma$-algebra
\[
  \mathcal B(\mathbb R^2)=\mathcal B(\mathbb R)\otimes\mathcal B(\mathbb R),
\]
generated by the measurable rectangles $A\times B$ with $A,B\in\mathcal B(\mathbb R)$. It is
also generated by the rectangles of the form
\[
  (-\infty,x]\times(-\infty,y]\quad(x,y\in\mathbb R),\qquad (a,b]\times(c,d]\quad(a<b,\ c<d),\qquad\dots
\]
the first of which are the ones used in probability (\cref{rem:pm-cdf-independence}).
\emph{Lebesgue measure on $\mathbb R^2$} is defined as the product measure
\[
  \underline m=m\otimes m,
\]
so that $\underline m(A\times B)=m(A)m(B)$ for all $A,B\in\mathcal B(\mathbb R)$; it exists
because $m$ is $\sigma$-finite. The notes state these facts without proof. The equality
$\mathcal B(\mathbb R^2)=\mathcal B(\mathbb R)\otimes\mathcal B(\mathbb R)$ is in
\citet[Section~1.2]{folland}; that the rectangles $(-\infty,x]\times(-\infty,y]$ generate
$\mathcal B(\mathbb R^2)$ is proved in \cref{rem:pm-cdf-independence}, and
the same argument works for the rectangles $(a,b]\times(c,d]$.\footnote{The handwritten notes
write $\underline m$ for the Lebesgue measure of the plane; we keep it. It is defined on
$\mathcal B(\mathbb R^2)$, not on its completion, which is all that the notes use.}
\end{example}

\section{The probabilistic case: random vectors}\label{sec:pm-random-vectors}

Throughout, $(\Omega,\mathcal F,\mathbb P)$ is a probability space.

\begin{definition}[Random vector]\label{def:pm-random-vector}
A \emph{real random vector}, or \emph{real vector random variable}, is a measurable function
\[
  \underline X:(\Omega,\mathcal F)\to(\mathbb R^N,\mathcal B(\mathbb R^N)),\qquad N\ge1 .
\]
For $N=1$ we recover a random variable (\cref{def:mf-random-variable}).
\end{definition}

If we denote by $X_i:\Omega\to\mathbb R$, $i=1,\dots,N$, the components of $\underline X$, we
can write
\[
  \underline X=(X_1,\dots,X_N),\qquad X_i=\pi_i\circ\underline X,
\]
where $\pi_i:\mathbb R^N\to\mathbb R$ is the $i$-th projection. Hence, if $\underline X$ is
measurable, so is every $X_i:(\Omega,\mathcal F)\to(\mathbb R,\mathcal B(\mathbb R))$
(\cref{prop:mf-composition}). Indeed
$\mathcal B(\mathbb R^N)=\mathcal B(\mathbb R)\otimes\dots\otimes\mathcal B(\mathbb R)$ and the
$\pi_i$ are measurable.\footnote{For $N$ factors, the product $\sigma$-algebra
$\mathcal A_1\otimes\dots\otimes\mathcal A_N$ is generated by the measurable rectangles
$A_1\times\dots\times A_N$, $A_i\in\mathcal A_i$, as in \cref{def:pm-product-sigma}, and
\cref{prop:pm-projections} holds with the same proof. The equality
$\mathcal B(\mathbb R^N)=\mathcal B(\mathbb R)\otimes\dots\otimes\mathcal B(\mathbb R)$ is in
\citet[Section~1.2]{folland}. The projection $\pi_i$ is measurable because
$\pi_i^{-1}(A)=\mathbb R\times\dots\times A\times\dots\times\mathbb R$ is a measurable
rectangle.}

Conversely, suppose that the $X_i$ are measurable. Take a measurable rectangle
$A_1\times\dots\times A_N\in\mathcal B(\mathbb R^N)$, that is, $A_i\in\mathcal B(\mathbb R)$ for
every $i=1,\dots,N$. Then
\[
\begin{aligned}
  \underline X^{-1}(A_1\times\dots\times A_N)
  &=\{\underline X\in A_1\times\dots\times A_N\}=\{(X_1,\dots,X_N)\in A_1\times\dots\times A_N\}\\
  &=\{X_1\in A_1,\dots,X_N\in A_N\}=\{X_1\in A_1\}\cap\dots\cap\{X_N\in A_N\}\\
  &=X_1^{-1}(A_1)\cap\dots\cap X_N^{-1}(A_N)\in\mathcal F,
\end{aligned}
\]
since each $X_i^{-1}(A_i)\in\mathcal F$. So $\underline X$ is measurable, because the
measurable rectangles generate $\mathcal B(\mathbb R^N)$ (\cref{prop:mf-generators}). This
gives an equivalent definition.

\begin{definition}[Random vector, equivalent form]\label{def:pm-random-vector-alt}
A real random vector is a function
$\underline X:(\Omega,\mathcal F)\to(\mathbb R^N,\mathcal B(\mathbb R^N))$ whose components
are real random variables.
\end{definition}

\begin{definition}[Joint and marginal laws]\label{def:pm-joint-law}
The \emph{joint law} of $\underline X=(X_1,\dots,X_N)$ is the image measure
$\mathbb P_{\underline X}$ (\cref{def:ld-image-measure}, \cref{def:ld-law}):
\[
  \mathbb P_{\underline X}(B)=\mathbb P(\underline X^{-1}(B))=\mathbb P(\underline X\in B)
  \qquad\forall B\in\mathcal B(\mathbb R^N).
\]
The laws $\mathbb P_{X_i}$ of the components are called the \emph{marginal laws}:
\[
  \mathbb P_{X_i}(A)=\mathbb P(X_i^{-1}(A))=\mathbb P(X_i\in A)\qquad\forall A\in\mathcal B(\mathbb R),\
  i=1,\dots,N .
\]
\end{definition}

\begin{definition}[Independent random variables]\label{def:pm-independent}
Let $(X_i)_{i\in I}$ be a family of real random variables, $I$ any index set. The random
variables $X_i$, $i\in I$, are \emph{independent} if the $\sigma$-algebras $\sigma(X_i)$,
$i\in I$ (\cref{def:mf-generated}), are independent (\cref{def:ms-independence-sigma}).
\end{definition}

From now on we take $N=2$ and write $Z=(X,Y)$. By definition, $X$ and $Y$ are independent,
$X\indep Y$, if and only if $\sigma(X)\indep\sigma(Y)$, where (\cref{prop:mf-sigma-f})
\[
  \sigma(X)=\{X^{-1}(A):A\in\mathcal B(\mathbb R)\},\qquad
  \sigma(Y)=\{Y^{-1}(B):B\in\mathcal B(\mathbb R)\}.
\]
Therefore
\[
\begin{aligned}
  X\indep Y&\iff X^{-1}(A)\indep Y^{-1}(B)\qquad\forall A,B\in\mathcal B(\mathbb R)\\
  &\iff\mathbb P\big(X^{-1}(A)\cap Y^{-1}(B)\big)=\mathbb P(X^{-1}(A))\,\mathbb P(Y^{-1}(B))
    \qquad\forall A,B\in\mathcal B(\mathbb R)\\
  &\iff\mathbb P(X\in A,\,Y\in B)=\mathbb P(X\in A)\,\mathbb P(Y\in B)
    \qquad\forall A,B\in\mathcal B(\mathbb R),
\end{aligned}
\]
and $\mathbb P(X\in A,\,Y\in B)=\mathbb P((X,Y)\in A\times B)$. That is,
\begin{equation}\label{eq:pm-rectangles}
  X\indep Y\iff\mathbb P_{(X,Y)}(A\times B)=\mathbb P_X(A)\,\mathbb P_Y(B)\qquad
  \forall A,B\in\mathcal B(\mathbb R),
\end{equation}
and by the uniqueness of the product measure we obtain the following theorem.

\begin{theorem}[Independence and the joint law]\label{thm:pm-independence-product}
Let $Z=(X,Y)$ be a real random vector. Then
\[
  X\indep Y\iff\mathbb P_{(X,Y)}=\mathbb P_X\otimes\mathbb P_Y :
\]
the joint law factorises into the marginal laws.
\end{theorem}

\begin{proof}
The laws $\mathbb P_X$ and $\mathbb P_Y$ are probability measures, hence $\sigma$-finite, so
$\mathbb P_X\otimes\mathbb P_Y$ is defined on
$\mathcal B(\mathbb R)\otimes\mathcal B(\mathbb R)=\mathcal B(\mathbb R^2)$
(\cref{thm:pm-product-measure}, \cref{rem:pm-borel-plane}). If $X\indep Y$,
\eqref{eq:pm-rectangles} says that the measure $\mathbb P_{(X,Y)}$ takes the value
$\mathbb P_X(A)\mathbb P_Y(B)$ on every rectangle $A\times B$, so it equals
$\mathbb P_X\otimes\mathbb P_Y$ by the uniqueness part of \cref{thm:pm-product-measure}.
Conversely, if $\mathbb P_{(X,Y)}=\mathbb P_X\otimes\mathbb P_Y$, then
$\mathbb P_{(X,Y)}(A\times B)=\mathbb P_X\otimes\mathbb P_Y(A\times B)=\mathbb P_X(A)\mathbb P_Y(B)$,
and \eqref{eq:pm-rectangles} gives $X\indep Y$.
\end{proof}

\begin{remark}[The joint distribution function]\label{rem:pm-cdf-independence}
Let $F_{(X,Y)}(x,y)=\mathbb P(X\le x,\,Y\le y)$ be the \emph{joint distribution function} of
$(X,Y)$, and $F_X$, $F_Y$ the distribution functions of $X$ and $Y$ (\cref{def:ld-cdf}). Then
\[
  X\indep Y\iff F_{(X,Y)}(x,y)=F_X(x)F_Y(y)\qquad\forall(x,y)\in\mathbb R^2 .
\]
Indeed, for every $(x,y)\in\mathbb R^2$,
\begin{equation}\label{eq:pm-cdf-identities}
\begin{aligned}
  F_{(X,Y)}(x,y)&=\mathbb P(X\le x,\,Y\le y)=\mathbb P\big(X\in(-\infty,x],\,Y\in(-\infty,y]\big)\\
  &=\mathbb P\big((X,Y)\in(-\infty,x]\times(-\infty,y]\big)
   =\mathbb P_{(X,Y)}\big((-\infty,x]\times(-\infty,y]\big),\\
  F_X(x)&=\mathbb P(X\le x)=\mathbb P\big(X\in(-\infty,x]\big)=\mathbb P_X\big((-\infty,x]\big),
  \qquad F_Y(y)=\mathbb P_Y\big((-\infty,y]\big).
\end{aligned}
\end{equation}
If $X\indep Y$, then $\mathbb P_{(X,Y)}=\mathbb P_X\otimes\mathbb P_Y$
(\cref{thm:pm-independence-product}), and
$F_{(X,Y)}(x,y)=\mathbb P_X\otimes\mathbb P_Y\big((-\infty,x]\times(-\infty,y]\big)=F_X(x)F_Y(y)$.

Conversely,\footnote{The handwritten notes justify this remark by the identities
\eqref{eq:pm-cdf-identities} alone. These give ``$\Rightarrow$'' at once. For ``$\Leftarrow$''
they only show that the two measures agree on the rectangles $(-\infty,x]\times(-\infty,y]$,
and one needs the uniqueness argument that follows, which the notes leave implicit; it is
added here.} assume $F_{(X,Y)}=F_XF_Y$. By \eqref{eq:pm-cdf-identities}, the two probability
measures $\mathbb P_{(X,Y)}$ and $\mathbb P_X\otimes\mathbb P_Y$ agree on the family
\[
  \mathcal Q=\big\{(-\infty,x]\times(-\infty,y]:\ x,y\in\mathbb R\big\}.
\]
This family is closed under finite intersections (a \emph{$\pi$-system}):
$\big((-\infty,x]\times(-\infty,y]\big)\cap\big((-\infty,x']\times(-\infty,y']\big)
=(-\infty,x\wedge x']\times(-\infty,y\wedge y']$. It generates $\mathcal B(\mathbb R^2)$:
on the one hand $\mathcal Q\subseteq\mathcal B(\mathbb R)\otimes\mathcal B(\mathbb R)$, so
$\sigma(\mathcal Q)\subseteq\mathcal B(\mathbb R^2)$; on the other hand
\[
  \pi_1^{-1}\big((-\infty,x]\big)=(-\infty,x]\times\mathbb R
  =\bigcup_{n\ge1}(-\infty,x]\times(-\infty,n]\in\sigma(\mathcal Q),
\]
and the half-lines $(-\infty,x]$ generate $\mathcal B(\mathbb R)$
(\cref{prop:sa-borel-generators}), so $\pi_1:(\mathbb R^2,\sigma(\mathcal Q))\to(\mathbb
R,\mathcal B(\mathbb R))$ is measurable (\cref{prop:mf-generators}); likewise $\pi_2$. Hence
$\sigma(\pi_1,\pi_2)\subseteq\sigma(\mathcal Q)$, and
$\sigma(\pi_1,\pi_2)=\mathcal B(\mathbb R)\otimes\mathcal B(\mathbb R)=\mathcal B(\mathbb R^2)$
by \cref{prop:pm-projections} and \cref{rem:pm-borel-plane}. Two probability measures that
agree on a $\pi$-system agree on the $\sigma$-algebra it generates: this is a consequence of
Dynkin's $\pi$--$\lambda$ theorem, \citet[Theorem~3.3; for $\sigma$-finite measures,
Theorem~10.3]{billingsley}. Therefore $\mathbb P_{(X,Y)}=\mathbb P_X\otimes\mathbb P_Y$ on
$\mathcal B(\mathbb R^2)$, and $X\indep Y$ by \cref{thm:pm-independence-product}. (The
uniqueness part of \cref{thm:pm-product-measure} is an instance of the same result, since the
measurable rectangles form a $\pi$-system.)
\end{remark}

\subsection{Discrete random vectors}\label{sec:pm-discrete}

A particular case. Let $Z=(X,Y)$. As for random variables (\cref{def:ld-discrete}), $Z$ is
\emph{discrete} if it takes finitely or countably many values, that is, if $Z(\Omega)$ is
finite or countable. Then
\[
  Z\text{ is discrete}\iff X\text{ and }Y\text{ are discrete random variables},
\]
and moreover $Z(\Omega)\subseteq X(\Omega)\times Y(\Omega)$. (If $X(\Omega)$ and $Y(\Omega)$
are at most countable, so is their product, which contains $Z(\Omega)$; conversely
$X(\Omega)=\pi_1(Z(\Omega))$ and $Y(\Omega)=\pi_2(Z(\Omega))$ are images of an at most
countable set.)

The joint law $\mathbb P_Z$ is then a discrete measure concentrated on $Z(\Omega)$
(\cref{def:ms-concentrated}):
\[
  \mathbb P_Z(B)=\sum_{(x,y)\in B\cap Z(\Omega)}f_Z(x,y)\qquad\forall B\in\mathcal B(\mathbb R^2),
\]
where
\[
  f_Z(x,y)=\mathbb P_Z(\{(x,y)\})=\mathbb P(Z=(x,y))=\mathbb P(X=x,\,Y=y)
\]
is the \emph{joint discrete density} of $(X,Y)=Z$. The \emph{marginal densities}
$f_X(x)=\mathbb P(X=x)$ and $f_Y(y)=\mathbb P(Y=y)$ are obtained from the joint density.

\begin{proposition}\label{prop:pm-discrete-independence}
Let $Z=(X,Y)$ be a discrete random vector with joint discrete density $f_Z$. Then
\begin{enumerate}[label=\arabic*)]
  \item the marginal densities are
    \[
    \begin{aligned}
      f_X(x)=\mathbb P(X=x)=\mathbb P_X(\{x\})&=\sum_{y\in Y(\Omega)}\mathbb P(X=x,\,Y=y)\\
      &=\sum_{y\in Y(\Omega)}f_Z(x,y)\qquad\forall x\in X(\Omega),
    \end{aligned}
    \]
    and $f_Y(y)=\mathbb P(Y=y)=\mathbb P_Y(\{y\})=\sum_{x\in X(\Omega)}f_Z(x,y)$ for all
    $y\in Y(\Omega)$;
  \item $X\indep Y\iff f_Z(x,y)=f_X(x)f_Y(y)$ for all $x,y\in\mathbb R$.
\end{enumerate}
\end{proposition}

The notes leave the proof of 1) and 2) as an exercise (\exref{ex:pm-discrete}). In 2),
$f_X(x)=\mathbb P(X=x)$ is defined for every real $x$, and it is $0$ outside $X(\Omega)$;
likewise for $f_Y$ and $f_Z$.

\begin{remark}[computational]\label{rem:pm-code-tables}
On finite spaces everything in this section is a computation with arrays.
\coderef{code/src/probnotes/product.py} writes a finite measure as a vector of masses, so that
the product measure of \cref{thm:pm-product-measure} is the outer product
(\texttt{product\_measure}), and checks on rectangles and on arbitrary sets that the two
iterated sums of \cref{prop:pm-sections} agree (\texttt{section\_measures} and
\texttt{product\_measure\_of\_set}). A joint discrete
density with finitely many values is a table: \texttt{marginals} computes the sums of
\cref{prop:pm-discrete-independence}, \texttt{is\_independent} tests the factorisation 2),
which for a table means having rank one, and \texttt{joint\_cdf} computes the joint
distribution function of \cref{rem:pm-cdf-independence}
(\exref{ex:pm-dice}).
\end{remark}

\section{Tonelli's and Fubini's theorems}\label{sec:pm-tonelli-fubini}

These are the theorems that reduce a double integral (with respect to the product measure)
to iterated simple integrals. They are the last topic of measure theory in the notes.

\begin{theorem}[Tonelli]\label{thm:pm-tonelli}
Let $(X_1,\mathcal A_1,\mu_1)$ and $(X_2,\mathcal A_2,\mu_2)$ be measure spaces with $\mu_1$
and $\mu_2$ $\sigma$-finite, and let $f:X_1\times X_2\to\overline{\mathbb R}$ be non-negative,
$f\ge0$, and measurable (with respect to $\mathcal A_1\otimes\mathcal A_2$). Then:
\begin{enumerate}[label=\arabic*)]
  \item for every $x\in X_1$, the function $f(x,\cdot):X_2\to\overline{\mathbb R}$,
    $y\mapsto f(x,y)$, is measurable and $\ge0$; for every $y\in X_2$, the function
    $f(\cdot,y):X_1\to\overline{\mathbb R}$, $x\mapsto f(x,y)$, is measurable and $\ge0$;
  \item the functions
    \[
      X_1\to\overline{\mathbb R},\quad x\mapsto\int_{X_2}f(x,y)\,\dd\mu_2(y),\qquad\qquad
      X_2\to\overline{\mathbb R},\quad y\mapsto\int_{X_1}f(x,y)\,\dd\mu_1(x)
    \]
    are measurable and $\ge0$;
  \item
    \[
    \begin{aligned}
      \int_{X_1}\Big(\int_{X_2}f(x,y)\,\dd\mu_2(y)\Big)\dd\mu_1(x)
      &=\int_{X_2}\Big(\int_{X_1}f(x,y)\,\dd\mu_1(x)\Big)\dd\mu_2(y)\\
      &=\int_{X_1\times X_2}f(x,y)\,\dd(\mu_1\otimes\mu_2)(x,y).
    \end{aligned}
    \]
\end{enumerate}
\end{theorem}

\begin{theorem}[Fubini]\label{thm:pm-fubini}
Let $(X_1,\mathcal A_1,\mu_1)$ and $(X_2,\mathcal A_2,\mu_2)$ be measure spaces with $\mu_1$
and $\mu_2$ $\sigma$-finite, and let $f:X_1\times X_2\to\overline{\mathbb R}$ be integrable
with respect to $\mu_1\otimes\mu_2$ (\cref{def:ig-integrable}). Then:
\begin{enumerate}[label=\arabic*)]
  \item for $\mu_1$-a.e.\ $x\in X_1$, the function $f(x,\cdot):X_2\to\overline{\mathbb R}$,
    $y\mapsto f(x,y)$, is integrable with respect to $\mu_2$; for $\mu_2$-a.e.\ $y\in X_2$, the
    function $f(\cdot,y):X_1\to\overline{\mathbb R}$, $x\mapsto f(x,y)$, is integrable with
    respect to $\mu_1$;
  \item the function
    \[
      X_1\to\mathbb R,\quad x\mapsto\int_{X_2}f(x,y)\,\dd\mu_2(y)
    \]
    is integrable with respect to $\mu_1$ (it is a function defined
    a.e.\footnote{It is defined outside the set of $x$ where $f(x,\cdot)$ is not
    $\mu_2$-integrable, a set in $\mathcal A_1$ by \cref{thm:pm-tonelli} applied to $|f|$, and
    $\mu_1$-negligible by 1); functions defined a.e.\ are integrated as in
    \cref{rem:ig-null-sets}. Added to the handwritten notes.}), and the function
    \[
      X_2\to\mathbb R,\quad y\mapsto\int_{X_1}f(x,y)\,\dd\mu_1(x)
    \]
    is integrable with respect to $\mu_2$ (a function defined a.e.);
  \item
    \[
    \begin{aligned}
      \int_{X_1}\Big(\int_{X_2}f(x,y)\,\dd\mu_2(y)\Big)\dd\mu_1(x)
      &=\int_{X_2}\Big(\int_{X_1}f(x,y)\,\dd\mu_1(x)\Big)\dd\mu_2(y)\\
      &=\int_{X_1\times X_2}f(x,y)\,\dd(\mu_1\otimes\mu_2)(x,y).
    \end{aligned}
    \]
\end{enumerate}
\end{theorem}

The notes state both theorems without proof. Tonelli's theorem is
\citet[Theorem~18.1(ii) and Theorem~18.3, first part]{billingsley} and
\citet[Theorem~8.8(a)]{rudin-rca}; Fubini's theorem is \citet[Theorem~18.3, second
part]{billingsley} and \citet[Theorem~8.8(c)]{rudin-rca}.\footnote{Added to the handwritten
notes: in practice one uses the two theorems together. Tonelli's theorem, applied to $|f|$,
tells whether $f$ is integrable, by computing $\int|f|\,\dd(\mu_1\otimes\mu_2)$ as an iterated
integral; if it is finite, Fubini's theorem allows one to exchange the order of integration
for $f$ itself (\cref{thm:pm-expectation-product} is an example).}

\begin{remark}[Double integrals]\label{rem:pm-double-integral}
On $(\mathbb R^2,\mathcal B(\mathbb R^2),\underline m)$, with $\underline m=m\otimes m$, we
write, with an abuse of notation,
\[
  \int_{\mathbb R^2}f\,\dd\underline m=\int_{\mathbb R^2}f(x,y)\,\dd(m\otimes m)(x,y)
  =\int_{-\infty}^{+\infty}\!\!\int_{-\infty}^{+\infty}f(x,y)\,\dd x\,\dd y .
\]
\end{remark}

\begin{remark}[computational]\label{rem:pm-code-fubini}
The integrability hypothesis of Fubini's theorem cannot be dropped. For
\[
  f(x,y)=\frac{x^2-y^2}{(x^2+y^2)^2}\qquad\text{on }(0,1]^2,
\]
both iterated integrals exist, but they are $\frac\pi4$ and $-\frac\pi4$
(\cref{fig:pm-fubini}): $f$ is not integrable on the square, and Tonelli's theorem applied to
$f^+$ and to $f^-$ gives $+\infty$ for both. The function \texttt{iterated\_integral} of
\coderef{code/src/probnotes/product.py} computes an iterated integral by the midpoint rule,
with separate resolutions for the inner and the outer integral. With the same grid in both variables the two
orders give the same finite double sum, here $0$ by antisymmetry; only when the inner integral
is resolved first, with many more points, do the two iterated integrals appear
(\exref{ex:pm-fubini-numerics}).\footnote{This remark and \cref{fig:pm-fubini} are added to the
handwritten notes. The example is classical; see also \citet[Counterexamples~8.9]{rudin-rca} for
counterexamples showing that every hypothesis of Fubini's theorem is needed.}
\end{remark}

\section{Absolutely continuous random vectors}\label{sec:pm-ac-vectors}

We return to the probabilistic case.

\begin{definition}[Absolutely continuous random vector]\label{def:pm-ac-vector}
A random vector $Z=(X,Y)$ is \emph{absolutely continuous} if
$\mathbb P_Z\ll\underline m=m\otimes m$ (\cref{def:ld-ac}). The Radon--Nikodym derivative
(\cref{thm:ld-radon-nikodym})
\[
  f_Z=\frac{\dd\mathbb P_Z}{\dd\underline m}\ge0
\]
is called the \emph{joint density} of $Z=(X,Y)$:
\[
  \mathbb P(Z\in B)=\mathbb P_Z(B)=\int_Bf_Z\,\dd\underline m\qquad\forall B\in\mathcal B(\mathbb R^2).
\]
In particular
$\int_{\mathbb R^2}f_Z\,\dd\underline m=\int_{-\infty}^{+\infty}\int_{-\infty}^{+\infty}f_Z(x,y)\,\dd x\,\dd y=1$.\footnote{Added
to the handwritten notes: the Radon--Nikodym theorem applies because $\mathbb P_Z$ is finite
and $\underline m$ is $\sigma$-finite; the joint density is determined up to
$\underline m$-a.e.\ equality, and can be taken with values in $[0,+\infty)$.}
\end{definition}

\begin{proposition}[Marginal densities]\label{prop:pm-marginal-density}
Let $Z=(X,Y)$ be an absolutely continuous random vector with density $f_Z$. Then $X$ and $Y$
are absolutely continuous random variables (\cref{def:ld-ac-rv}), and
\[
  f_X(x)=\int_{-\infty}^{+\infty}f_Z(x,y)\,\dd y,\qquad
  f_Y(y)=\int_{-\infty}^{+\infty}f_Z(x,y)\,\dd x
\]
for $m$-a.e.\ $x\in\mathbb R$ and $m$-a.e.\ $y\in\mathbb R$.\footnote{The handwritten notes write
``for every $x\in\mathbb R$'' and ``for every $y\in\mathbb R$''. The integrals can be
$+\infty$ on an $m$-negligible set (for instance if $f_Z$ is changed on a line, which is
$\underline m$-negligible), and a density is only determined up to a.e.\ equality. The
formulas define densities of $X$ and $Y$ once they are set to $0$ where they are infinite;
corrected here.}
\end{proposition}

\begin{proof}
Let $A\in\mathcal B(\mathbb R)$. Then
\[
\begin{aligned}
  \mathbb P_X(A)&=\mathbb P(X\in A)=\mathbb P(X\in A,\,Y\in\mathbb R)=\mathbb P\big((X,Y)\in A\times\mathbb R\big)
  =\int_{A\times\mathbb R}f_Z\,\dd\underline m\\
  &=\int_{\mathbb R^2}\1_A(x)f_Z(x,y)\,\dd x\,\dd y
  =\int_{\mathbb R}\1_A(x)\Big(\int_{\mathbb R}f_Z(x,y)\,\dd y\Big)\dd x,
\end{aligned}
\]
by Tonelli's theorem (\cref{thm:pm-tonelli}) applied to the non-negative measurable function
$(x,y)\mapsto\1_A(x)f_Z(x,y)$. By part 2 of the same theorem,
$g(x)=\int_{\mathbb R}f_Z(x,y)\,\dd y$ is a measurable function with values in $[0,+\infty]$, and
for $A=\mathbb R$ the computation gives $\int_{\mathbb R}g\,\dd m=1$; hence $g<+\infty$ a.e.\
(\cref{prop:in-ae}). Let $f_X=g$ where $g<+\infty$ and $f_X=0$ elsewhere. Then $f_X\ge0$ is
integrable and
\[
  \mathbb P_X(A)=\int_Af_X(x)\,\dd x\qquad\forall A\in\mathcal B(\mathbb R),
\]
so $\mathbb P_X\ll m$, with $f_X=\frac{\dd\mathbb P_X}{\dd m}$ and
$f_X(x)=\int_{-\infty}^{+\infty}f_Z(x,y)\,\dd y$ for a.e.\ $x$. The argument for $f_Y$ is the
same.
\end{proof}

\begin{proposition}[Independence and densities]\label{thm:pm-density-independence}
Let $Z=(X,Y)$ be absolutely continuous, and let $f_X$ and $f_Y$ be the densities of $X$ and
$Y$ (\cref{prop:pm-marginal-density}). Then
\[
  X\indep Y\iff f_Z(x,y)=f_X(x)f_Y(y)\quad\underline m\text{-a.e.}
\]
\end{proposition}

\begin{proof}
For $A,B\in\mathcal B(\mathbb R)$ we compare
\[
  \underbrace{\mathbb P(Z\in A\times B)}_{=\,\mathbb P_Z(A\times B)}=\mathbb P(X\in A,\,Y\in B)
  =\int_{A\times B}f_Z\,\dd\underline m\tag{1}
\]
with
\[
  \underbrace{\mathbb P(X\in A)\,\mathbb P(Y\in B)}_{=\,\mathbb P_X(A)\,\mathbb P_Y(B)}
  =\int_Af_X\,\dd m\cdot\int_Bf_Y\,\dd m=\int_{A\times B}f_Xf_Y\,\dd\underline m,\tag{2}
\]
where $f_Xf_Y$ denotes the function $(x,y)\mapsto f_X(x)f_Y(y)$, and the last equality is
Tonelli's theorem applied to $(x,y)\mapsto\1_A(x)f_X(x)\,\1_B(y)f_Y(y)\ge0$.

``$\Leftarrow$''. Let $f_Z=f_Xf_Y$ a.e. Then (1)$=$(2) for all $A,B\in\mathcal B(\mathbb R)$,
since functions that are equal a.e.\ have the same integral over every set
(\cref{prop:in-ae}). So
$\mathbb P_Z(A\times B)=\mathbb P_X(A)\mathbb P_Y(B)$ for all $A,B$, and $X\indep Y$ by
\eqref{eq:pm-rectangles}.

``$\Rightarrow$''. Let $X\indep Y$. Then (1)$=$(2) for all $A,B\in\mathcal B(\mathbb R)$.
Define a measure on $\mathcal B(\mathbb R^2)$ by (\cref{prop:ld-density-measure})
\[
  \mu(C)=\int_Cf_Xf_Y\,\dd\underline m\qquad\forall C\in\mathcal B(\mathbb R^2).
\]
Then $\mu(A\times B)=\mathbb P_X(A)\mathbb P_Y(B)=\mathbb P_Z(A\times B)$ for all
$A,B\in\mathcal B(\mathbb R)$, and by the uniqueness of the product measure
(\cref{thm:pm-product-measure}; both $\mu$ and $\mathbb P_Z$ equal
$\mathbb P_X\otimes\mathbb P_Y$), $\mu(C)=\mathbb P_Z(C)$ for every
$C\in\mathcal B(\mathbb R^2)$. In particular
\[
  \int_Cf_Xf_Y\,\dd\underline m=\int_Cf_Z\,\dd\underline m\qquad\forall C\in\mathcal B(\mathbb R^2),
\]
and since $f_Xf_Y$ and $f_Z$ are integrable (both integrals equal $1$ for $C=\mathbb R^2$),
$f_Xf_Y=f_Z$ a.e.\ (\cref{prop:ig-ae}).
\end{proof}

\begin{remark}[A factorised density]\label{rem:pm-factorisation}
If $f_Z(x,y)=f_1(x)f_2(y)$ $\underline m$-a.e., for some non-negative Borel functions $f_1$ and
$f_2$, then $X\indep Y$.\footnote{Added to the handwritten notes: it is not necessary to know
in advance that $f_1$ and $f_2$ are the marginal densities.} Indeed, by Tonelli's theorem
\[
  1=\int_{\mathbb R^2}f_Z(x,y)\,\dd x\,\dd y
  =\underbrace{\int_{-\infty}^{+\infty}f_1(x)\,\dd x}_{=\,c}\ \cdot\
   \underbrace{\int_{-\infty}^{+\infty}f_2(y)\,\dd y}_{=\,1/c},
\]
so that $0<c<+\infty$. For a.e.\ $x\in\mathbb R$ (\cref{prop:pm-marginal-density})
\[
  f_X(x)=\int_{-\infty}^{+\infty}f_Z(x,y)\,\dd y=f_1(x)\int_{-\infty}^{+\infty}f_2(y)\,\dd y=f_1(x)\cdot\frac1c,
\]
and for a.e.\ $y\in\mathbb R$
\[
  f_Y(y)=\int_{-\infty}^{+\infty}f_Z(x,y)\,\dd x=f_2(y)\int_{-\infty}^{+\infty}f_1(x)\,\dd x=f_2(y)\cdot c .
\]
Moreover
\[
  f_Z(x,y)\overset{\text{a.e.}}{=}f_1(x)f_2(y)=\Big(f_1(x)\cdot\frac1c\Big)\big(c\cdot f_2(y)\big)=f_X(x)f_Y(y),
\]
and $X\indep Y$ by \cref{thm:pm-density-independence}.\footnote{The handwritten notes write
``for every $x\in\mathbb R$'' in the two computations of the marginal densities. Since
$f_Z=f_1f_2$ only a.e., the inner integrals agree only for a.e.\ $x$ (by
\cref{rem:pm-null-sections} applied to $\{f_Z\ne f_1f_2\}$), and for a.e.\ $y$; corrected
here.}
\end{remark}

\section{An application of Fubini--Tonelli}\label{sec:pm-application}

Let $g:\mathbb R^2\to\mathbb R$ be measurable and such that
$g(X,Y)=g\circ(X,Y)\in L^1(\Omega,\mathcal F,\mathbb P)$.\footnote{Added to the handwritten
notes: $g(X,Y)$ is a random variable, by \cref{prop:mf-composition}.} Then, with $Z=(X,Y)$ (\cref{thm:ld-image-integral},
\cref{def:ld-expectation}),
\[
  \mathbb E[g(X,Y)]=\int_\Omega g(X,Y)\,\dd\mathbb P=\int_{\mathbb R^2}g(x,y)\,\dd\mathbb P_Z(x,y).
\]
In particular, if $X,Y\in L^2$, then $XY\in L^1$ (\cref{cor:ld-cauchy-schwarz}; this is the
case $g(x,y)=xy$), and
\[
  \mathbb E[XY]=\int_\Omega XY\,\dd\mathbb P=\int_{\mathbb R^2}xy\,\dd\mathbb P_Z(x,y).
\]

\begin{proposition}\label{thm:pm-expectation-product}
Let $X,Y\in L^1$ with $X\indep Y$. Then $XY\in L^1$ \emph{(important!)} and
\[
  \mathbb E[XY]=\mathbb E[X]\,\mathbb E[Y],
\]
so that $\operatorname{cov}(X,Y)=0$.\footnote{The notes do not define the covariance here; we
recall it. For $X,Y\in L^1$ with $XY\in L^1$, $\operatorname{cov}(X,Y)=\mathbb E\big[(X-\mathbb
E[X])(Y-\mathbb E[Y])\big]=\mathbb E[XY]-\mathbb E[X]\,\mathbb E[Y]$; $X$ and $Y$ are
\emph{uncorrelated} if $\operatorname{cov}(X,Y)=0$. Added to the handwritten notes.}
\end{proposition}

\begin{proof}
$X,Y\in L^1$ implies $|X|,|Y|\in L^1$ (\cref{prop:ig-abs}). Then
\[
\begin{aligned}
  +\infty>\mathbb E[|X|]\,\mathbb E[|Y|]
  &=\int_\Omega|X|\,\dd\mathbb P\cdot\int_\Omega|Y|\,\dd\mathbb P
   =\int_{\mathbb R}|x|\,\dd\mathbb P_X(x)\cdot\int_{\mathbb R}|y|\,\dd\mathbb P_Y(y)\\
  &\overset{(a)}{=}\int_{\mathbb R^2}|xy|\,\dd(\mathbb P_X\otimes\mathbb P_Y)(x,y)
   \overset{(b)}{=}\int_{\mathbb R^2}|xy|\,\dd\mathbb P_Z(x,y)
   =\int_\Omega|XY|\,\dd\mathbb P=\mathbb E[|XY|],
\end{aligned}
\]
where (a) is Tonelli's theorem (\cref{thm:pm-tonelli}) applied to
$f(x,y)=|x|\,|y|=|xy|\ge0$, and (b) holds because $X\indep Y$, that is,
$\mathbb P_X\otimes\mathbb P_Y=\mathbb P_Z$ (\cref{thm:pm-independence-product}); the other
equalities are \cref{thm:ld-image-integral}. Hence
\[
  \mathbb E[|XY|]=\mathbb E[|X|]\,\mathbb E[|Y|]<+\infty\ \Longrightarrow\ |XY|\in L^1
  \ \Longrightarrow\ XY\in L^1 .
\]
So $(x,y)\mapsto xy$ is integrable with respect to $\mathbb P_X\otimes\mathbb P_Y=\mathbb P_Z$,
and we can use Fubini's theorem (\cref{thm:pm-fubini}): repeating the computation without
absolute values gives $\mathbb E[XY]=\mathbb E[X]\,\mathbb E[Y]$. The notes leave this last step
as an exercise (\exref{ex:pm-expectation-product}).
\end{proof}

\begin{remark}[computational]\label{rem:pm-code-uncorrelated}
The converse of \cref{thm:pm-expectation-product} is false: uncorrelated random variables need
not be independent. \Cref{fig:pm-dependence} shows four pairs with zero covariance, of which
only the first is independent. The \emph{CDF gap} under each panel is the largest value of
$|F_{(X,Y)}(x,y)-F_X(x)F_Y(y)|$ over a grid of points, computed from $10^5$ samples by
\texttt{empirical\_cdf\_defect} in \coderef{code/src/probnotes/product.py}. By
\cref{rem:pm-cdf-independence} it tends to $0$ for independent pairs, so a positive limit
proves dependence (on a fixed grid, a limit $0$ does not prove independence). For
$(X,X^2)$ with $X$ uniform on $(-1,1)$ the gap is $\frac18$, attained at $(-\frac12,\frac14)$:
$\mathbb P(X\le-\frac12,\,X^2\le\frac14)=0$, whereas $F_X(-\frac12)F_{X^2}(\frac14)=\frac14\cdot\frac12$.
The pair $(\cos T,\sin T)$, with $T$ uniform on $(0,2\pi)$, lies on the unit circle, an
$\underline m$-negligible set: it has no joint density although each component has one
(\exref{ex:pm-uncorrelated}).
\end{remark}
